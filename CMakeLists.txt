cmake_minimum_required(VERSION 3.24)
project(qemu_3dfx_winhelper VERSION 0.3.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "Implement the Host interface for this platform first; core has no WinUI dependency.")
endif()
add_library(helper_core STATIC src/core/Types.cpp src/core/Qmp.cpp src/core/Manager.cpp src/platform/WindowsHost.cpp)
target_include_directories(helper_core PUBLIC src third_party)
target_compile_definitions(helper_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(helper_core PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(helper_core PUBLIC ws2_32 bcrypt)
add_executable(qemu-helper-cli src/cli/main.cpp)
target_link_libraries(qemu-helper-cli PRIVATE helper_core)
target_compile_options(qemu-helper-cli PRIVATE /utf-8)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE helper_core shell32)
target_compile_options(core_tests PRIVATE /utf-8)
add_test(NAME core_tests COMMAND core_tests)
add_executable(qemu-3dfx-winhelper WIN32 src/launcher/main.cpp src/assets/WinHelper.rc src/winui/app.manifest)
target_include_directories(qemu-3dfx-winhelper PRIVATE src src/assets)
target_compile_definitions(qemu-3dfx-winhelper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(qemu-3dfx-winhelper PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
target_link_libraries(qemu-3dfx-winhelper PRIVATE shell32 user32)
set_property(TARGET qemu-3dfx-winhelper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
